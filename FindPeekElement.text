//Find Peak Element
//[1,2,1,3,5,6,4] l=0 , r = 6 , mid = 3 

//  travel towards the highest number to reach peek  
// if mid > mid +1 go towards right 
// else go towards left 

class Solution {
    public int findPeakElement(int[] nums) {
        int left = 0;
        int right = nums.length-1;
        while(left < right){ // [1,2,1,3,5,6,4] l=0 , r = 6 , mid = 3  // [1]
            int mid = left+ (right-left) /2;
            if(nums[mid] > nums[mid+1]){ 
                right = mid;
            } 
            else left = mid+1; 
        }
        return left;
    }
}
