//Comparing the mid with a minVlaue and storing it 
// then navigating towards the unsorted array to identify the min element 
class Solution {
    public int findMin(int[] nums) {
        int left = 0;
        int right = nums.length -1;
        int minVal = nums[left];
        while(left <= right){
            int mid = left +(right-left) /2;
            minVal = Math.min(minVal,nums[mid]);
            if(nums[left] > nums[right] && nums[mid] >= nums[left]){
                left = mid+1;
            }else {
                right = mid-1;
            }
        }
        return minVal;
    }
}
